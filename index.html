import pygame import sys # 1. تهيئة مكتبة Pygame pygame.init() pygame.font.init() # 2. إعدادات الشاشة والألوان SCREEN_WIDTH = 600 SCREEN_HEIGHT = 700 screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT)) pygame.display.set_caption("The Angry Teddy Bear Game 🧸") WHITE = (255, 255, 255) BLACK = (0, 0, 0) BEAR_BROWN = (139, 69, 19) MUZZLE_CREAM = (245, 222, 179) KEFFIYEH_BLUE = (0, 100, 150) RED = (200, 0, 0) # إعداد الخطوط للكتابة العربية/الإنجليزية (استخدم خط يدعم العربية في جهازك إذا لزم الأمر) try: font = pygame.font.SysFont("Arial", 24) dialogue_font = pygame.font.SysFont("Arial", 28, bold=True) except: font = pygame.font.Font(None, 36) dialogue_font = pygame.font.Font(None, 40) # 3. حالات اللعبة (States) # "GREETING" -> يرحب بك # "ANGRY" -> عصب وطلع المسدس current_state = "GREETING" text_bubble = "أهلاً بك! أنا أتكلم معك الآن 👋" # 4. دالة رسم الدب بناءً على حالته def draw_bear(state): # رسم الخلفية والملابس (الشماغ الأزرق) screen.fill((230, 240, 255)) # خلفية الغرفة # الجسد pygame.draw.ellipse(screen, BEAR_BROWN, (150, 400, 300, 300)) # الأذنين pygame.draw.circle(screen, BEAR_BROWN, (180, 220), 45) pygame.draw.circle(screen, BEAR_BROWN, (420, 220), 45) # الرأس (موقع اللمس الأساسي) bear_head_rect = pygame.Rect(160, 220, 280, 260) pygame.draw.ellipse(screen, BEAR_BROWN, bear_head_rect) # الشماغ/الغطاء الأزرق فوق الرأس (نفس شكل الصورة) pygame.draw.arc(screen, KEFFIYEH_BLUE, (155, 200, 290, 100), 0, 3.14, 25) # منطقة الفم والأنف (البيج) pygame.draw.ellipse(screen, MUZZLE_CREAM, (230, 340, 140, 110)) # الأنف البني pygame.draw.ellipse(screen, (90, 45, 10), (275, 350, 50, 35)) # العيون والتعابير حسب الحالة if state == "GREETING": # عيون عادية pygame.draw.circle(screen, BLACK, (260, 300), 12) pygame.draw.circle(screen, BLACK, (340, 300), 12) # ابتسامة سعيدة pygame.draw.arc(screen, BLACK, (260, 380, 80, 40), 3.14, 0, 3) elif state == "ANGRY": # عيون غاضبة (مائلة) pygame.draw.circle(screen, RED, (260, 300), 14) pygame.draw.circle(screen, RED, (340, 300), 14) # حواجب غاضبة pygame.draw.line(screen, BLACK, (240, 270), (280, 290), 5) pygame.draw.line(screen, BLACK, (360, 270), (320, 290), 5) # فم حزين/معصب pygame.draw.arc(screen, BLACK, (260, 400, 80, 40), 0, 3.14, 3) # 🔫 رسم مسدس بسيط في يد الدب # اليد (المخالب) pygame.draw.circle(screen, BEAR_BROWN, (140, 480), 35) # المسدس (مستطيلات رمادية) pygame.draw.rect(screen, (80, 80, 80), (70, 440, 70, 25)) # السبطانة pygame.draw.rect(screen, (50, 50, 50), (110, 440, 25, 50)) # المقبض return bear_head_rect # 5. دالة رسم فقاعة الكلام def draw_speech_bubble(text, is_angry): bubble_color = (255, 230, 230) if is_angry else WHITE border_color = RED if is_angry else BLACK # صندوق الكلام pygame.draw.rect(screen, bubble_color, (50, 50, 500, 80), border_radius=15) pygame.draw.rect(screen, border_color, (50, 50, 500, 80), width=3, border_radius=15) # نص الكلام (ملاحظة: لإنياك النصوص العربية بشكل مثالي قد تحتاج مكتبة bidi، هنا وضعنا نص توضيحي) text_surface = dialogue_font.render(text, True, BLACK) text_rect = text_surface.get_rect(center=(300, 90)) screen.blit(text_surface, text_rect) # 6. حلقة اللعبة الرئيسية (Game Loop) running = True while running: # الحصول على إحداثيات الماوس mouse_pos = pygame.mouse.get_pos() # رسم الدب والحصول على منطقة الرأس الحساسة للمس bear_rect = draw_bear(current_state) draw_speech_bubble(text_bubble, current_state == "ANGRY") # إدارة الأحداث (Events) for event in pygame.event.get(): if event.type == pygame.QUIT: running = False pygame.quit() sys.exit() # التحقق من لمس (الضغط بالماوس) على الدب if event.type == pygame.MOUSEBUTTONDOWN: if bear_rect.collidepoint(mouse_pos): if current_state == "GREETING": current_state = "ANGRY" text_bubble = "لقد حذرتك! الآن ستواجه العواقب! 😡🔫" else: # إذا ضغطت في مكان آخر يرجع لحالته الطبيعية current_state = "GREETING" text_bubble = "أهلاً بك! أنا أتكلم معك الآن 👋" pygame.display.flip()
